-- DATABASE: SQLite

-- STEP 1: Create the patients table

CREATE TABLE IF NOT EXISTS patients (
    patient_id INTEGER PRIMARY KEY,
    patient_name TEXT,
    age INTEGER,
    gender TEXT,
    department TEXT,
    bill_amount REAL,
    satisfaction_score INTEGER
);

-- STEP 3: Display all patients

SELECT *
FROM patients;

-- STEP 4: Calculate total hospital revenue

SELECT SUM(bill_amount) AS total_revenue
FROM patients;

-- STEP 5: Calculate average patient age

SELECT AVG(age) AS average_patient_age
FROM patients;

-- STEP 6: Calculate revenue by department

SELECT
    department,
    SUM(bill_amount) AS total_revenue
FROM patients
GROUP BY department
ORDER BY total_revenue DESC;

-- STEP 7: Calculate average satisfaction by department

SELECT
    department,
    ROUND(AVG(satisfaction_score), 2) AS avg_satisfaction
FROM patients
GROUP BY department
ORDER BY avg_satisfaction DESC;

-- STEP 8: Identify patients with bills above 20000

SELECT
    patient_id,
    department,
    bill_amount
FROM patients
WHERE bill_amount > 20000
ORDER BY bill_amount DESC;
